# PlanMe Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build PlanMe — an Electron + React desktop app that ingests Claude Code project plans as `plan.json` files and renders them as interactive, traversable flowchart graphs with a notebook aesthetic.

**Architecture:** Three subsystems share a `plan.json` schema contract: (1) a `planme` Claude Code skill that extracts plans from conversation history and writes `plan.md`; (2) a Node.js parser agent that converts `plan.md` to `plan.json` via the Claude API; (3) an Electron + React app that watches `~/.planme/plans/`, auto-ingests `plan.json` files, and renders them as interactive graphs using React Flow. The app is the primary deliverable; the skill and parser are its input pipeline.

**Tech Stack:** Electron, React 18, Vite, electron-vite, React Flow (@xyflow/react v12), Zustand 5, chokidar 3, Anthropic SDK, uuid, Vitest 2, @testing-library/react 16

```js
// src/schema/plan.js
const VALID_STATUSES = ['pending', 'in_progress', 'completed']

function validateNode(node) {
  if (!node.id || typeof node.id !== 'string') return false
  if (!node.title || typeof node.title !== 'string') return false
  if (!node.description || typeof node.description !== 'string') return false
  if (!Array.isArray(node.checklist)) return false
  if (!node.verification || typeof node.verification !== 'string') return false
  if (!VALID_STATUSES.includes(node.status)) return false
  if (!node.position || typeof node.position.x !== 'number') return false
  if (!Array.isArray(node.dependsOn)) return false
  if (!Array.isArray(node.children)) return false
  return true
}

export function validatePlan(plan) {
  if (!plan) return false
  if (!plan.version || !plan.id || !plan.title || !plan.createdAt) return false
  if (!plan.overview || !plan.overview.description || !Array.isArray(plan.overview.techStack)) return false
  if (!Array.isArray(plan.nodes)) return false
  return plan.nodes.every(validateNode)
}

export function isNodeBlocked(node, allNodes) {
  if (!node.dependsOn || node.dependsOn.length === 0) return false
  return node.dependsOn.some(depId => {
    const dep = allNodes.find(n => n.id === depId)
    return !dep || dep.status !== 'completed'
  })
}
```

```js
// parser/layout.js
const NODE_WIDTH = 240
const NODE_HEIGHT = 130
const H_GAP = 60
const V_GAP = 80

function computeLayout(nodes) {
  const rowMap = {}

  function getRow(nodeId) {
    if (rowMap[nodeId] !== undefined) return rowMap[nodeId]
    const node = nodes.find(n => n.id === nodeId)
    if (!node || node.dependsOn.length === 0) {
      rowMap[nodeId] = 0
      return 0
    }
    rowMap[nodeId] = Math.max(...node.dependsOn.map(getRow)) + 1
    return rowMap[nodeId]
  }

  nodes.forEach(n => getRow(n.id))

  const rowIndexes = {}
  return nodes.map(node => {
    const row = rowMap[node.id]
    const idx = rowIndexes[row] || 0
    rowIndexes[row] = idx + 1
    return {
      ...node,
      position: {
        x: idx * (NODE_WIDTH + H_GAP),
        y: row * (NODE_HEIGHT + V_GAP)
      }
    }
  })
}

module.exports = { computeLayout }
```

```js
// parser/index.js
const fs = require('fs')
const Anthropic = require('@anthropic-ai/sdk')
const { computeLayout } = require('./layout.js')

const SCHEMA_PROMPT = `You are a plan parser. Convert the given plan.md into a JSON object matching this exact schema. Return ONLY valid JSON — no markdown fences, no explanation.

{
  "version": "1.0",
  "id": "<uuid-v4>",
  "title": "string",
  "createdAt": "<ISO 8601 now>",
  "overview": {
    "description": "string",
    "techStack": [{ "layer": "string", "technology": "string" }]
  },
  "nodes": [
    {
      "id": "<uuid-v4>",
      "title": "string",
      "description": "string",
      "checklist": [{ "id": "<uuid-v4>", "text": "string", "checked": false }],
      "verification": "string",
      "status": "pending",
      "position": { "x": 0, "y": 0 },
      "dependsOn": ["<id of another node in this same array>"],
      "children": [
        {
          "id": "<uuid-v4>",
          "title": "string",
          "description": "string",
          "checklist": [{ "id": "<uuid-v4>", "text": "string", "checked": false }],
          "verification": "string",
          "status": "pending",
          "position": { "x": 0, "y": 0 },
          "dependsOn": ["<id of sibling child node only>"],
          "children": []
        }
      ]
    }
  ]
}

Rules:
- Generate a UUID v4 for every id field. Use crypto.randomUUID() format: xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx
- dependsOn at the top level references ids of other top-level nodes only
- dependsOn inside children references ids of sibling children only — never parent ids
- children cannot have children (max 2 levels)
- All status: "pending", all checked: false, all positions: {"x":0,"y":0} (layout computed separately)
- Extract ALL nodes from the plan — do not summarize or omit steps`

async function parsePlan(mdPath) {
  if (!process.env.ANTHROPIC_API_KEY) {
    console.error('Error: ANTHROPIC_API_KEY environment variable is not set.')
    process.exit(1)
  }

  const mdContent = fs.readFileSync(mdPath, 'utf-8')
  const titleMatch = mdContent.match(/^# (.+)$/m)
  const title = titleMatch ? titleMatch[1] : 'Unknown plan'
  const client = new Anthropic()

  let plan
  try {
    const message = await client.messages.create({
      model: 'claude-sonnet-4-6',
      max_tokens: 8096,
      messages: [{ role: 'user', content: `${SCHEMA_PROMPT}\n\nPlan to convert:\n\n${mdContent}` }]
    })
    const raw = message.content[0].text.trim()
    plan = JSON.parse(raw)
  } catch (err) {
    const errorPath = mdPath.replace(/\.md$/, '.error.json')
    fs.writeFileSync(errorPath, JSON.stringify({
      title,
      error: err.message,
      timestamp: new Date().toISOString()
    }, null, 2))
    console.error(`Parser failed. Error written to ${errorPath}`)
    console.error(err.message)
    process.exit(1)
  }

  // Apply layout
  plan.nodes = computeLayout(plan.nodes)
  plan.nodes = plan.nodes.map(node => ({
    ...node,
    children: node.children.length > 0 ? computeLayout(node.children) : []
  }))

  const jsonPath = mdPath.replace(/\.md$/, '.json')
  fs.writeFileSync(jsonPath, JSON.stringify(plan, null, 2))
  console.log(`Plan written to ${jsonPath}`)
}

const mdPath = process.argv[2]
if (!mdPath) {
  console.error('Usage: node parser/index.js <path-to-plan.md>')
  process.exit(1)
}

parsePlan(mdPath)
```

```js
// electron/main/index.js
const { app, BrowserWindow, ipcMain } = require('electron')
const path = require('path')
const fs = require('fs')
const os = require('os')
const chokidar = require('chokidar')

const PLANS_DIR = path.join(os.homedir(), '.planme', 'plans')

let mainWindow
// Maps planId -> { plan, filePath }
const loadedPlans = new Map()

function createWindow() {
  mainWindow = new BrowserWindow({
    width: 1400,
    height: 900,
    webPreferences: {
      preload: path.join(__dirname, '../preload/index.js'),
      contextIsolation: true,
      nodeIntegration: false,
    },
    titleBarStyle: 'hiddenInset',
    backgroundColor: '#F5F0E8',
  })

  if (process.env['ELECTRON_RENDERER_URL']) {
    mainWindow.loadURL(process.env['ELECTRON_RENDERER_URL'])
  } else {
    mainWindow.loadFile(path.join(__dirname, '../../out/renderer/index.html'))
  }
}

function sendToRenderer(channel, data) {
  mainWindow?.webContents.send(channel, data)
}

function loadJsonFile(filePath) {
  if (!filePath.endsWith('.json') || filePath.endsWith('.error.json')) return
  try {
    const raw = fs.readFileSync(filePath, 'utf-8')
    const plan = JSON.parse(raw)
    if (!plan.id) return

    const isNew = !loadedPlans.has(plan.id)
    loadedPlans.set(plan.id, { plan, filePath })

    if (isNew) {
      sendToRenderer('plan:loaded', plan)
    } else {
      sendToRenderer('plan:prompt-reload', { planId: plan.id })
    }
  } catch (err) {
    console.error(`Failed to load ${filePath}:`, err.message)
    sendToRenderer('plan:error', { filePath, error: err.message })
  }
}

function loadErrorFile(filePath) {
  if (!filePath.endsWith('.error.json')) return
  try {
    const raw = fs.readFileSync(filePath, 'utf-8')
    const data = JSON.parse(raw)
    sendToRenderer('plan:error', { filePath, ...data })
  } catch {}
}

app.whenReady().then(() => {
  createWindow()

  // Load plans present on startup
  fs.readdirSync(PLANS_DIR).forEach(file => {
    const full = path.join(PLANS_DIR, file)
    loadJsonFile(full)
    loadErrorFile(full)
  })

  // Watch for changes
  const watcher = chokidar.watch(PLANS_DIR, { ignoreInitial: true })
  watcher.on('add', (fp) => { loadJsonFile(fp); loadErrorFile(fp) })
  watcher.on('change', (fp) => {
    if (fp.endsWith('.json') && !fp.endsWith('.error.json')) loadJsonFile(fp)
  })

  // Write plan back to disk
  ipcMain.handle('plan:write', async (_e, plan) => {
    const entry = loadedPlans.get(plan.id)
    if (entry) {
      fs.writeFileSync(entry.filePath, JSON.stringify(plan, null, 2))
      entry.plan = plan
    }
    return { success: true }
  })

  // Confirm reload after external change
  ipcMain.handle('plan:confirm-reload', async (_e, planId) => {
    const entry = loadedPlans.get(planId)
    if (entry) {
      const plan = JSON.parse(fs.readFileSync(entry.filePath, 'utf-8'))
      entry.plan = plan
      sendToRenderer('plan:changed', plan)
    }
    return { success: true }
  })
})

app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') app.quit()
})

app.on('activate', () => {
  if (BrowserWindow.getAllWindows().length === 0) createWindow()
})
```

```js
// src/store/planStore.js
import { create } from 'zustand'

function mutatePlan(plans, planId, mutator) {
  return plans.map(p => p.id === planId ? mutator(p) : p)
}

function mutateNode(plan, nodeId, mutator) {
  return { ...plan, nodes: plan.nodes.map(n => n.id === nodeId ? mutator(n) : n) }
}

export const usePlanStore = create((set, get) => ({
  plans: [],
  activePlanId: null,
  activeTab: 'graph',
  errors: [],

  addPlan: (plan) => set(state => ({
    plans: state.plans.some(p => p.id === plan.id) ? state.plans : [...state.plans, plan],
    activePlanId: state.activePlanId ?? plan.id,
  })),

  updatePlan: (plan) => set(state => ({
    plans: mutatePlan(state.plans, plan.id, () => plan),
  })),

  setActivePlan: (planId) => set({ activePlanId: planId }),

  setActiveTab: (tab) => set({ activeTab: tab }),

  addError: (error) => set(state => ({ errors: [...state.errors, error] })),

  toggleChecklistItem: (planId, nodeId, itemId) => {
    const plan = get().plans.find(p => p.id === planId)
    if (!plan) return
    const updated = mutateNode(plan, nodeId, node => ({
      ...node,
      checklist: node.checklist.map(item =>
        item.id === itemId ? { ...item, checked: !item.checked } : item
      ),
    }))
    set(state => ({ plans: mutatePlan(state.plans, planId, () => updated) }))
    window.planme?.writePlan(updated)
  },

  updateNodeStatus: (planId, nodeId, status) => {
    const plan = get().plans.find(p => p.id === planId)
    if (!plan) return
    const updated = mutateNode(plan, nodeId, node => ({ ...node, status }))
    set(state => ({ plans: mutatePlan(state.plans, planId, () => updated) }))
    window.planme?.writePlan(updated)
  },

  updateNodePosition: (planId, nodeId, position) => {
    const plan = get().plans.find(p => p.id === planId)
    if (!plan) return
    const updated = mutateNode(plan, nodeId, node => ({ ...node, position }))
    set(state => ({ plans: mutatePlan(state.plans, planId, () => updated) }))
    window.planme?.writePlan(updated)
  },
}))
```

```jsx
import { Handle, Position } from '@xyflow/react'
import { usePlanStore } from '../store/planStore.js'
import '../styles/plannode.css'

export default function PlanNode({ data }) {
  const { planId, node, isBlocked, expandedNodeId, onExpand, onCollapse, onDrillIn } = data
  const expanded = expandedNodeId === node.id
  const dimmed = expandedNodeId !== null && !expanded
  const { toggleChecklistItem, updateNodeStatus } = usePlanStore()

  const classNames = ['plan-node',
    node.status === 'completed' && 'completed',
    isBlocked && 'blocked',
    expanded && 'expanded',
    dimmed && 'dimmed',
  ].filter(Boolean).join(' ')

  function handleClick() {
    expanded ? onCollapse() : onExpand(node.id)
  }

  function handleStatusCycle(e) {
    e.stopPropagation()
    const next = { pending: 'in_progress', in_progress: 'completed', completed: 'pending' }
    updateNodeStatus(planId, node.id, next[node.status])
  }

  return (
    <>
      <Handle type="target" position={Position.Top} />
      <div className={classNames} onClick={handleClick}>
        <div className="node-header">
          {node.status === 'in_progress' && <span className="status-dot in-progress" />}
          <span className="node-title">{node.title}</span>
          <button className="status-btn" onClick={handleStatusCycle} title={`Status: ${node.status}`}>
            {node.status === 'completed' ? '✓' : node.status === 'in_progress' ? '◐' : '○'}
          </button>
        </div>

        {expanded && (
          <div className="node-body" onClick={e => e.stopPropagation()}>
            <p className="node-description">{node.description}</p>

            {node.checklist.length > 0 && (
              <ul className="node-checklist">
                {node.checklist.map(item => (
                  <li key={item.id}>
                    <label>
                      <input
                        type="checkbox"
                        checked={item.checked}
                        onChange={() => toggleChecklistItem(planId, node.id, item.id)}
                      />
                      {item.text}
                    </label>
                  </li>
                ))}
              </ul>
            )}

            {node.verification && (
              <div className="node-verification">
                <span className="verification-label">Done when:</span>
                <span>{node.verification}</span>
              </div>
            )}

            {node.children?.length > 0 && (
              <button className="drill-in-btn" onClick={() => onDrillIn?.(node)}>
                Go deeper →
              </button>
            )}
          </div>
        )}
      </div>
      <Handle type="source" position={Position.Bottom} />
    </>
  )
}
```

```jsx
// src/components/PlanCanvas.jsx
import { useState } from 'react'
import { ReactFlow, Background, Controls } from '@xyflow/react'
import '@xyflow/react/dist/style.css'
import PlanNode from './PlanNode.jsx'
import SubCanvas from './SubCanvas.jsx'
import { usePlanStore } from '../store/planStore.js'
import { isNodeBlocked } from '../schema/plan.js'
import '../styles/canvas.css'

const nodeTypes = { planNode: PlanNode }

export function deriveEdges(nodes) {
  const edges = []
  nodes.forEach(node => {
    node.dependsOn.forEach(depId => {
      edges.push({
        id: `${depId}->${node.id}`,
        source: depId,
        target: node.id,
        type: 'smoothstep',
        style: { stroke: '#1B2D4F', strokeWidth: 1.5, opacity: 0.5 },
      })
    })
  })
  return edges
}

export default function PlanCanvas({ plan }) {
  const { updateNodePosition } = usePlanStore()
  const [subGraphNode, setSubGraphNode] = useState(null)
  const [expandedNodeId, setExpandedNodeId] = useState(null)

  const rfNodes = plan.nodes.map(node => ({
    id: node.id,
    type: 'planNode',
    position: node.position,
    data: {
      planId: plan.id,
      node,
      isBlocked: isNodeBlocked(node, plan.nodes),
      expandedNodeId,
      onExpand: (id) => setExpandedNodeId(id),
      onCollapse: () => setExpandedNodeId(null),
      onDrillIn: (n) => setSubGraphNode(n),
    },
  }))

  if (subGraphNode) {
    return <SubCanvas plan={plan} parentNode={subGraphNode} onBack={() => setSubGraphNode(null)} />
  }

  return (
    <div className="canvas-wrapper">
      {expandedNodeId && (
        <div className="canvas-backdrop" onClick={() => setExpandedNodeId(null)} />
      )}
      <ReactFlow
        nodes={rfNodes}
        edges={deriveEdges(plan.nodes)}
        nodeTypes={nodeTypes}
        onNodeDragStop={(_e, rfNode) => updateNodePosition(plan.id, rfNode.id, rfNode.position)}
        fitView
      >
        <Background color="#D9D0C0" gap={24} size={1} />
        <Controls />
      </ReactFlow>
    </div>
  )
}
```

```jsx
import { useState } from 'react'
import { ReactFlow, Background, Controls } from '@xyflow/react'

import { deriveEdges } from './PlanCanvas.jsx'

export default function SubCanvas({ plan, parentNode, onBack }) {

    },
  }))

  return (
    <div className="canvas-wrapper">
      <div className="breadcrumb">
        <button className="breadcrumb-link" onClick={onBack}>{plan.title}</button>
        <span className="breadcrumb-sep"> → </span>
```

```markdown
# PlanMe

Extract the implementation plan from this conversation, write it to disk, and trigger the parser agent to convert it to a graph.

## Steps

1. Extract a structured plan from the current conversation history.
2. Write `~/.planme/plans/<slugified-title>.md` using the template below.
3. Run: `node ~/.planme/parser/index.js ~/.planme/plans/<slugified-title>.md`
4. The PlanMe desktop app auto-detects the new plan.json and adds it to the sidebar.

## Prerequisites

- `ANTHROPIC_API_KEY` must be set in the environment.
- The parser must be installed at `~/.planme/parser/` (copy from the PlanMe repo's `parser/` directory).
- The PlanMe app should be running to see the plan appear immediately.

## plan.md Template

Use this exact format:

```
# <Plan Title>

**Description:** <What this plan is and why — 1-3 sentences>

**Tech Stack:**
- <Layer>: <Technology>

## <Node Title>
**Description:** <What this step accomplishes>

**Checklist:**
- [ ] <Sub-task>
- [ ] <Sub-task>

**Verification:** You know you're done when <criterion>.

**Depends on:** <Node Title> (or "none")

### <Child Node Title> (only for multi-step sub-processes within a node)
**Description:** <Brief description>

**Checklist:**
- [ ] <Sub-task>

**Verification:** You know you're done when <criterion>.

**Depends on:** <Sibling Child Node Title> (or "none")
```

## Rules

- Extract ALL nodes — do not summarize or omit steps.
- Dependencies reference node titles exactly as written.
- Child nodes are for multi-step sub-processes within a single top-level node only.
- Children cannot have children (2-level max).
- Slugify the title for the filename: lowercase, spaces to hyphens, alphanumeric only.
```
